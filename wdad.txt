<script>
function updateOrder() {
    let coffeeSelect = document.getElementById("coffee");
    let coffeeName = coffeeSelect.options[coffeeSelect.selectedIndex].text;
    let coffeePrice = parseInt(coffeeSelect.value);

    let qty = parseInt(document.getElementById("qty").value);

    let addOns = document.querySelectorAll(".addons input:checked");
    let addOnList = [];
    let addOnTotal = 0;

    addOns.forEach(function(item) {
        addOnList.push(item.parentNode.textContent.trim());
        addOnTotal += parseInt(item.value);
    });

    let total = (coffeePrice * qty) + addOnTotal;

    // Update summary
    document.getElementById("summary").innerHTML =
        "Coffee: " + coffeeName + "<br>" +
        "Quantity: " + qty + "<br>" +
        "Add-ons: " + (addOnList.length ? addOnList.join(", ") : "None");

    // Update total
    document.getElementById("total").innerText = total;
}

function placeOrder() {
    alert("Order placed successfully! ☕");
}
