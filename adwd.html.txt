<script>
function calculateTotal() {
  let coffee = document.getElementById("coffee");
  let coffeePrice = parseFloat(coffee.value);
  let coffeeName = coffee.options[coffee.selectedIndex].text;

  let addons = document.querySelectorAll(".addon:checked");
  let addonTotal = 0;
  let addonNames = [];

  addons.forEach(addon => {
    addonTotal += parseFloat(addon.value);
    addonNames.push(addon.parentElement.textContent.trim());
  });

  let quantity = parseInt(document.getElementById("quantity").value);

  let total = (coffeePrice + addonTotal) * quantity;

  document.getElementById("summary").innerHTML =
    "<b>Coffee:</b> " + coffeeName + "<br>" +
    "<b>Add-ons:</b> " + (addonNames.length ? addonNames.join(", ") : "None") + "<br>" +
    "<b>Quantity:</b> " + quantity;

  document.getElementById("total").innerText = "Total: $" + total;
}
</script>
